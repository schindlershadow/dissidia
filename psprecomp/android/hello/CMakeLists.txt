cmake_minimum_required(VERSION 3.20)

set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)

add_library(main SHARED hello.c)
target_link_libraries(main PRIVATE SDL2::SDL2 GLESv3 log android)
